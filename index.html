import cv2 import collections import time # ==== 初期設定 ==== delay_seconds = 5.0 # スタートディレイ（秒） record_seconds = 10.0 # 録画時間（秒） CAMERA_ID = 0 # カメラデバイスID # ================== def draw_ui(img, status_text, color, delay_val, record_val, timer_val=None): """画面上に現在のステータスやUIを表示する補助関数""" h, w, _ = img.shape # 上部ステータスバーの背景（半透明黒） overlay = img.copy() cv2.rectangle(overlay, (0, 0), (w, 90), (0, 0, 0), -1) cv2.addWeighted(overlay, 0.6, img, 0.4, 0, img) # ステータス表示 cv2.putText(img, f"STATUS: {status_text}", (20, 35), cv2.FONT_HERSHEY_SIMPLEX, 0.9, color, 2, cv2.LINE_AA) # カウントダウン表示（必要な時のみ） if timer_val is not None: cv2.putText(img, f"{timer_val:.1f}s", (w - 140, 50), cv2.FONT_HERSHEY_SIMPLEX, 1.3, (255, 255, 255), 3, cv2.LINE_AA) # 下部設定ガイドの背景 cv2.rectangle(overlay, (0, h - 70), (w, h), (0, 0, 0), -1) cv2.addWeighted(overlay, 0.6, img, 0.4, 0, img) # 設定値と操作ガイド表示 info_text = f"[<-/->] Delay: {delay_val:.1f}s | [^/v] Rec: {record_val:.1f}s" guide_text = "[SPACE]: Start | [P]: Play/Stop | [Q]: Quit" cv2.putText(img, info_text, (20, h - 40), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 229, 255), 1, cv2.LINE_AA) cv2.putText(img, guide_text, (20, h - 15), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (200, 200, 200), 1, cv2.LINE_AA) def run_off_training_camera(): global delay_seconds, record_seconds cap = cv2.VideoCapture(CAMERA_ID) if not cap.isOpened(): print("エラー: カメラが見つかりません。") return fps = cap.get(cv2.CAP_PROP_FPS) if fps <= 0: fps = 30.0 state = "READY" # READY, WAITING, RECORDING, REVIEW state_start_time = 0 recorded_frames = [] review_index = 0 is_playing_review = False print("🚀 オフトレカメラ Pro 起動") while True: ret, frame = cap.read() if not ret: break # 鏡のように左右反転 frame = cv2.flip(frame, 1) current_time = time.time() display_frame = frame.copy() # --- 状態管理 --- if state == "READY": draw_ui(display_frame, "READY (Press SPACE)", (0, 229, 255), delay_seconds, record_seconds) elif state == "WAITING": elapsed = current_time - state_start_time remaining = delay_seconds - elapsed if remaining <= 0: state = "RECORDING" state_start_time = time.time() recorded_frames = [] else: draw_ui(display_frame, "START IN...", (0, 255, 255), delay_seconds, record_seconds, remaining) elif state == "RECORDING": elapsed = current_time - state_start_time remaining = record_seconds - elapsed recorded_frames.append(frame.copy()) if remaining <= 0: state = "REVIEW" review_index = 0 is_playing_review = False # 自動再生せず停止待機（省電力UI） else: draw_ui(display_frame, "● REC", (0, 0, 255), delay_seconds, record_seconds, remaining) elif state == "REVIEW": if len(recorded_frames) > 0: if is_playing_review: display_frame = recorded_frames[review_index].copy() review_index = (review_index + 1) % len(recorded_frames) status_lbl = "REPLAYING (Press P to Pause)" status_color = (76, 217, 100) else: display_frame = recorded_frames[0].copy() # サムネイル（1コマ目）固定 status_lbl = "PAUSED (Press P to Play)" status_color = (200, 200, 200) draw_ui(display_frame, status_lbl, status_color, delay_seconds, record_seconds) cv2.imshow("Off-Training Camera Pro", display_frame) # --- キー入力操作 (UI/UX) --- key = cv2.waitKey(int(1000 / fps)) & 0xFF if key == ord('q'): break # 設定変更（READY状態またはREVIEW状態で変更可能） if state in ["READY", "REVIEW"]: if key == 81 or key == 2: # 左矢印キー (<-) delay_seconds = max(0.0, delay_seconds - 1.0) elif key == 83 or key == 3: # 右矢印キー (->) delay_seconds += 1.0 elif key == 84 or key == 1: # 下矢印キー (v) record_seconds = max(1.0, record_seconds - 1.0) elif key == 82 or key == 0: # 上矢印キー (^) record_seconds += 1.0 # スペースキー：撮影スタート（いつでも復帰） if key == 32: # SPACE state = "WAITING" if delay_seconds > 0 else "RECORDING" state_start_time = time.time() recorded_frames = [] # Pキー：確認用の再生/停止切り替え if key == ord('p') and state == "REVIEW": is_playing_review = not is_playing_review cap.release() cv2.destroyAllWindows() if __name__ == "__main__": run_off_training_camera()
