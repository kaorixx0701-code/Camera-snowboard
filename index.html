import cv2 import time # 初期設定 delay_seconds = 3 record_seconds = 10 CAMERA_ID = 0 state = "READY" # READY, WAITING, RECORDING, REVIEW state_start_time = 0 recorded_frames = [] review_index = 0 is_playing = False # ボタンのクリック処理 def mouse_callback(event, x, y, flags, param): global delay_seconds, record_seconds, state, state_start_time, recorded_frames, is_playing if event == cv2.EVENT_LBUTTONDOWN: # Delay - ボタン if 20 <= x <= 60 and 80 <= y <= 115: delay_seconds = max(0, delay_seconds - 1) # Delay + ボタン elif 160 <= x <= 200 and 80 <= y <= 115: delay_seconds += 1 # Rec - ボタン elif 220 <= x <= 260 and 80 <= y <= 115: record_seconds = max(1, record_seconds - 1) # Rec + ボタン elif 360 <= x <= 400 and 80 <= y <= 115: record_seconds += 1 # メインアクションボタン（スタート / 再生 / 停止） elif 20 <= x <= 220 and 130 <= y <= 180: if state in ["READY", "REVIEW"]: state = "WAITING" if delay_seconds > 0 else "RECORDING" state_start_time = time.time() recorded_frames = [] is_playing = False # 再生/一時停止ボタン（REVIEW時のみ） elif 240 <= x <= 400 and 130 <= y <= 180 and state == "REVIEW": is_playing = not is_playing def draw_button(img, text, pt1, pt2, bg_color, text_color=(255, 255, 255)): """ボタン描画用関数""" cv2.rectangle(img, pt1, pt2, bg_color, -1) cv2.rectangle(img, pt1, pt2, (200, 200, 200), 1) # テキスト配置位置の調整 text_size = cv2.getTextSize(text, cv2.FONT_HERSHEY_SIMPLEX, 0.6, 2)[0] tx = pt1[0] + (pt2[0] - pt1[0] - text_size[0]) // 2 ty = pt1[1] + (pt2[1] - pt1[1] + text_size[1]) // 2 cv2.putText(img, text, (tx, ty), cv2.FONT_HERSHEY_SIMPLEX, 0.6, text_color, 2, cv2.LINE_AA) def main(): global state, state_start_time, recorded_frames, review_index, is_playing cap = cv2.VideoCapture(CAMERA_ID) if not cap.isOpened(): print("エラー: カメラが起動できません。") return fps = cap.get(cv2.CAP_PROP_FPS) if fps <= 0: fps = 30.0 window_name = "Off-Training Camera Pro" cv2.namedWindow(window_name) cv2.setMouseCallback(window_name, mouse_callback) while True: ret, frame = cap.read() if not ret: break frame = cv2.flip(frame, 1) h, w, _ = frame.shape current_time = time.time() display = frame.copy() # 状態更新 if state == "WAITING": elapsed = current_time - state_start_time if elapsed >= delay_seconds: state = "RECORDING" state_start_time = time.time() recorded_frames = [] elif state == "RECORDING": elapsed = current_time - state_start_time recorded_frames.append(frame.copy()) if elapsed >= record_seconds: state = "REVIEW" review_index = 0 is_playing = False # 録画直後は電池節約のため静止 elif state == "REVIEW": if len(recorded_frames) > 0: if is_playing: display = recorded_frames[review_index].copy() review_index = (review_index + 1) % len(recorded_frames) else: display = recorded_frames[0].copy() # --- UIパネル（上部コントロールエリア） --- overlay = display.copy() cv2.rectangle(overlay, (0, 0), (w, 195), (20, 20, 20), -1) cv2.addWeighted(overlay, 0.7, display, 0.3, 0, display) # ステータス表示 status_colors = { "READY": (255, 229, 0), "WAITING": (0, 255, 255), "RECORDING": (50, 50, 255), "REVIEW": (100, 217, 76) } status_text = state if state == "WAITING": rem = max(0.0, delay_seconds - (current_time - state_start_time)) status_text = f"START IN {rem:.1f}s" elif state == "RECORDING": rem = max(0.0, record_seconds - (current_time - state_start_time)) status_text = f"REC {rem:.1f}s" elif state == "REVIEW": status_text = "PAUSED (Tap Play)" if not is_playing else "PLAYING" cv2.putText(display, f"STATUS: {status_text}", (20, 40), cv2.FONT_HERSHEY_SIMPLEX, 0.8, status_colors[state], 2, cv2.LINE_AA) # Delay 設定コントロール cv2.putText(display, f"Delay: {delay_seconds}s", (75, 103), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 1) draw_button(display, "-", (20, 80), (60, 115), (50, 50, 50)) draw_button(display, "+", (160, 80), (200, 115), (50, 50, 50)) # Record 設定コントロール cv2.putText(display, f"Rec: {record_seconds}s", (280, 103), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 1) draw_button(display, "-", (220, 80), (260, 115), (50, 50, 50)) draw_button(display, "+", (360, 80), (400, 115), (50, 50, 50)) # アクションボタン btn_label = "START REC" if state in ["READY", "REVIEW"] else "CANCEL" btn_color = (0, 150, 255) if state in ["READY", "REVIEW"] else (50, 50, 200) draw_button(display, btn_label, (20, 130), (220, 180), btn_color) # リプレイボタン（REVIEW時のみ） if state == "REVIEW": p_label = "PAUSE" if is_playing else "PLAY" p_color = (100, 100, 100) if is_playing else (76, 217, 100) draw_button(display, p_label, (240, 130), (400, 180), p_color, (0, 0, 0) if not is_playing else (255,255,255)) cv2.imshow(window_name, display) # 'q' キーで終了 if cv2.waitKey(int(1000 / fps)) & 0xFF == ord('q'): break cap.release() cv2.destroyAllWindows() if __name__ == "__main__": main()
